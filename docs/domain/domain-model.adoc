= Domain Model

== User
* id - PK
* username
* surname
* name
* email
* password
* role
* status
* department_id - FK
* manager_email
* benefit
* notes
* created_at
* last_modified_at

== Department
* id - PK
* name
* manager_id - FK
* created_at
* last_modified_at

== AssetCategory
* id - PK
* name
* description
* booking_period
* approval
* created_at
* last_modified_at

== Asset
* id - PK
* name
* category_id- FK
* description
* code
* status
* location
* created_at
* last_modified_at

== Booking
* id - PK
* user_id - FK
* asset_id - FK
* status
* booking_start_time
* booking_end_time
* notes
* created_at
* last_modified_at
